const { useState, useEffect, useRef, useCallback } = React;

let store = {};

function fakeLogin(email){ return new Promise(r=>setTimeout(()=>r({email}),700)); }
function fakeUpload(file,title,onProg){
  return new Promise(r=>{
    let p=0;
    const iv=setInterval(()=>{
      p=Math.min(p+18,100); onProg(p);
      if(p>=100){
        clearInterval(iv);
        const id='m'+Date.now();
        const m={id,title:title||(file?file.name.replace(/\.[^.]+$/,''):'Untitled'),status:'queued',transcript:null,createdAt:new Date().toISOString()};
        store[id]=m;
        setTimeout(()=>{if(store[id])store[id].status='transcribing';},1200);
        setTimeout(()=>{if(store[id]){store[id].status='transcribed';store[id].transcript='Host: Thanks everyone for joining. Let us get started with the agenda.\nSpeaker 1: I have completed the assigned tasks from last week and will share the update shortly.\nHost: Great. Any blockers we should know about before we move forward?\nSpeaker 2: No blockers on my end. Everything is on track for the deadline.\nHost: Perfect. Let us move to the next item. We will wrap up in the next ten minutes.\nSpeaker 1: Sounds good. I will send the summary notes after this call.\nHost: Thank you everyone. See you next time.';}},5500);
        r(m);
      }
    },120);
  });
}
function fakeGet(id){return new Promise(r=>setTimeout(()=>r({...store[id]}),150));}
function fakeList(){return new Promise(r=>setTimeout(()=>r(Object.values(store).reverse()),150));}
function startPoll(id,cb,ms=2000){
  let stopped=false;
  const tick=async()=>{if(stopped)return;const m=await fakeGet(id);cb(m);if(m.status==='transcribed'||m.status==='failed')return;if(!stopped)setTimeout(tick,ms);};
  tick();return()=>{stopped=true;};
}
